\section{Introducción}

\begin{frame}{Contexto y Problemática}
  \begin{block}{¿De qué se trata el proyecto?}
    Simular el comportamiento de **cargas eléctricas electrostáticas** en un dominio 2D para encontrar la configuración de **mínima energía potencial** (estado más estable).
  \end{block}

  \pause

  \begin{alertblock}{Problemática}
    \begin{itemize}
      \item La energía potencial es una función de alta dimensionalidad
      \item No existe una solución analítica general para N > 2
      \item Se necesita un método numérico eficiente para explorar el espacio de configuraciones
    \end{itemize}
  \end{alertblock}
\end{frame}

\begin{frame}{Objetivos del Proyecto}
  \begin{exampleblock}{Objetivo General}
    Desarrollar un simulador computacional de cargas eléctricas que utilice el método Monte Carlo a temperatura cero (T=0) para minimizar la energía potencial electrostática.
  \end{exampleblock}

  \pause

  \begin{block}{Objetivos Específicos}
    \begin{enumerate}
      \item Implementar el cálculo de la energía potencial electrostática
      \item Desarrollar el algoritmo de Monte Carlo a T=0
      \item Crear visualizaciones científicas claras y profesionales
      \item Analizar los resultados y la convergencia del sistema
    \end{enumerate}
  \end{block}
\end{frame}

\section{Marco Teórico}

\begin{frame}{Ley de Coulomb y Energía Potencial}
  \begin{block}{Ley de Coulomb \cite{carga-molina2024}}
    La fuerza entre dos cargas puntuales es:
    \[
    \vec{F}_{12} = k_e \frac{q_1 q_2}{r_{12}^2} \hat{r}_{12}
    \]
    donde \( k_e = \frac{1}{4\pi\epsilon_0} \approx 8.988 \times 10^9 \, \text{N·m}^2/\text{C}^2 \).
  \end{block}

  \pause

  \begin{block}{Energía Potencial Electrostática}
    La energía total del sistema es la suma sobre todos los pares:
    \[
    U = k_e \sum_{i < j} \frac{q_i q_j}{|\vec{r}_i - \vec{r}_j|}
    \]
  \end{block}
\end{frame}

\begin{frame}{Método Monte Carlo a T=0 \cite{proyecto-molina2024}}
  \begin{alertblock}{¿Cómo funciona?}
    Es un algoritmo de **descenso de gradiente estocástico**:
    \begin{enumerate}
      \item Seleccionar una partícula al azar
      \item Moverla una pequeña distancia aleatoria
      \item Calcular el cambio de energía \( \Delta U \)
      \item \textbf{Aceptar} si \( \Delta U < 0 \) (energía disminuye)
      \item \textbf{Rechazar} si \( \Delta U \geq 0 \) (energía aumenta)
    \end{enumerate}
  \end{alertblock}

  \pause

  \begin{block}{Ventaja}
    No requiere calcular gradientes — solo compara energías!
  \end{block}
\end{frame}
